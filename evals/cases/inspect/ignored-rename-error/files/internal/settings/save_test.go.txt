package settings

import (
	"path/filepath"
	"testing"
)

func TestSaveLoad(t *testing.T) {
	p := filepath.Join(t.TempDir(), "s.json")
	want := Settings{Currency: "EUR", PageSize: 20}
	if err := Save(p, want); err != nil {
		t.Fatal(err)
	}
	if got, err := Load(p); err != nil || got != want {
		t.Fatalf("got %+v %v", got, err)
	}
}
