package metrics

import (
	"net/http/httptest"
	"strings"
	"sync"
	"testing"
)

func TestHandlerAfterThreeIncs(t *testing.T) {
	c := NewCounter()
	for i := 0; i < 3; i++ {
		c.Inc("orders")
	}
	w := httptest.NewRecorder()
	c.Handler(w, httptest.NewRequest("GET", "/metrics", nil))
	if got := strings.TrimSpace(w.Body.String()); got != `{"orders":3}` {
		t.Fatalf("GET /metrics = %s", got)
	}
}

// Run with go test -race: the handler reads while 100 goroutines increment.
func TestHandlerWhileIncrementing(t *testing.T) {
	c := NewCounter()
	var wg sync.WaitGroup
	for i := 0; i < 100; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			c.Inc("orders")
		}()
	}
	for i := 0; i < 10; i++ {
		c.Handler(httptest.NewRecorder(), httptest.NewRequest("GET", "/metrics", nil))
	}
	wg.Wait()
	if got := c.Snapshot()["orders"]; got != 100 {
		t.Fatalf("orders = %d after 100 increments", got)
	}
}
