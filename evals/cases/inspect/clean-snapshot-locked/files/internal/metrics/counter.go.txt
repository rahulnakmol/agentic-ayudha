package metrics

import (
	"encoding/json"
	"net/http"
	"sync"
)

// Counter counts named events. It is safe for concurrent use.
type Counter struct {
	mu sync.Mutex
	n  map[string]int
}

// NewCounter returns an empty counter.
func NewCounter() *Counter { return &Counter{n: map[string]int{}} }

// Inc adds one to name.
func (c *Counter) Inc(name string) {
	c.mu.Lock()
	c.n[name]++
	c.mu.Unlock()
}

// Snapshot returns a copy of every count, taken under the lock, so later
// increments never change it.
func (c *Counter) Snapshot() map[string]int {
	c.mu.Lock()
	defer c.mu.Unlock()
	out := make(map[string]int, len(c.n))
	for k, v := range c.n {
		out[k] = v
	}
	return out
}

// Handler serves GET /metrics as a JSON object of counts.
func (c *Counter) Handler(w http.ResponseWriter, _ *http.Request) {
	w.Header().Set("Content-Type", "application/json")
	_ = json.NewEncoder(w).Encode(c.Snapshot()) // a write error means the client left; there is no one to tell
}
