package settings

import "fmt"

// DefaultPageSize is the page size of settings that do not set one.
const DefaultPageSize = 20

// Settings are the shop's preferences.
type Settings struct {
	Currency string `json:"currency"`
	PageSize int    `json:"page_size"`
}

// Normalize fills in defaults and checks ranges: a page size of 0 becomes
// DefaultPageSize, and one outside 1-100 is an error.
func (s *Settings) Normalize() error {
	if s.PageSize == 0 {
		s.PageSize = DefaultPageSize
	}
	if s.PageSize < 1 || s.PageSize > 100 {
		return fmt.Errorf("settings: page_size %d is outside 1-100", s.PageSize)
	}
	return nil
}
